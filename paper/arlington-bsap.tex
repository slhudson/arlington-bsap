% Heading conventions, set once for the whole report. A part (Part A to Part
% C) opens a page. A section (A.1) is numbered within its part and starts a

% A part-level line is bold and flush left, numbered ("Part A.") or not (the
% Executive Summary, Works Cited, Legal Authorities, each co-equal with a
% Part and not under one), with one fixed gap above it. A section-level line
% (A.1, indented under the part-level line above it) carries none of its
% own, so every part-level line sits the same distance below whatever came
% before it --- another part-level line or a run of sections --- instead of
% that distance depending on which kind of line happened to precede it.
\titlecontents{part}[0pt]{\addvspace{10pt}\bfseries}{Part \thecontentslabel.\hspace{0.5em}}{}{\hfill\contentspage}
\titlecontents{section}[3.6em]{}{\contentslabel{2.4em}}{\hspace*{-3.6em}}{\hfill\contentspage}
\renewcommand{\thepart}{\Alph{part}}
\titleformat{\part}{\normalfont\normalsize\bfseries}{Part \thepart.}{0.5em}{}
% One scheme, set once, for every level (Sally, 5 October 2026): one unit,
% a line of body text (\baselineskip), and every heading's space above is
% 1 or 2 of it, never a point value chosen by eye. A subsection and a
% run-in head, which already share the same italic weight, get a line; a
% section and a part, bold, get two, so where sections run on with no page
% break between them (Parts B, C and D), a section still reads as a bigger
% break than a subsection. Three lines read as excessive for a section
% (Sally, 5 October 2026). Space below is one line throughout, so a
% heading sits close to its own text and the hierarchy reads in the space
% above, not below.
\titlespacing*{\part}{0pt}{2\baselineskip}{\baselineskip}
% Sections are numbered within their part: A.1, A.2, ..., C.1.
\counterwithin*{section}{part}
\renewcommand{\thesection}{\thepart.\arabic{section}}
\titlespacing*{\section}{0pt}{2\baselineskip}{\baselineskip}
\titlespacing*{\subsection}{0pt}{\baselineskip}{\baselineskip}
% Third level, used inside Demographic Representation: unnumbered, italic, run on the
% way a subsection is, a little closer to its text.
\titleformat{\subsubsection}{\normalfont\normalsize\itshape}{}{0em}{}
\titlespacing*{\subsubsection}{0pt}{\baselineskip}{\baselineskip}

% Revision marks (Sally, 7 and 8 October 2026), on part lines too. The first line of every
% section file records who has touched it, "% revised: SH 7 October 2026"
% (several entries separated by "; "), Claude being one of the authors as CC.
% \sectioninput reads that line, then \inputs the file, and the contents page
% stamps the latest entry small and grey beside each section's entry, "SH 7
% Oct": who touched the draft last. The file is the one place the record
% lives; code/revisions.py lists the coverage and holds the rule that a
% person, not only CC, has revised a section before a draft copy is kept.
\usepackage{xcolor}

\newread\rev@file
\def\rev@abbr#1#2#3#4\relax{#1#2#3}  % October -> Oct
\def\rev@m@January{1}\def\rev@m@February{2}\def\rev@m@March{3}\def\rev@m@April{4}
\def\rev@m@May{5}\def\rev@m@June{6}\def\rev@m@July{7}\def\rev@m@August{8}
\def\rev@m@September{9}\def\rev@m@October{10}\def\rev@m@November{11}\def\rev@m@December{12}
\newcount\rev@key\newcount\rev@best
% One revision, "SH 7 October 2026": the latest seen so far is kept, as "SH 7 Oct".
\def\rev@try#1 #2 #3 #4\relax{%
  \rev@key=\numexpr#4*10000+\csname rev@m@#3\endcsname*100+#2\relax
  \ifnum\rev@key>\rev@best
    \global\rev@best=\rev@key
    \xdef\rev@bestitem{#1 #2 \rev@abbr#3\relax}%
  \fi}
\def\rev@scan#1; #2\relax{\rev@try#1\relax\ifx\relax#2\relax\else\rev@scan#2\relax\fi}
\def\rev@parse#1: #2\relax#3\relax{\def\rev@status{#2}}
% Reads the first line of a file into \rev@status.
\newcommand{\rev@read}[1]{%
  \openin\rev@file=#1.tex
  \ifeof\rev@file\PackageError{revisions}{No section file #1.tex}{}\fi
  \readline\rev@file to \rev@line
  \closein\rev@file
  \expandafter\rev@parse\rev@line\relax: \relax
  \ifx\rev@status\@empty\PackageError{revisions}{#1.tex has no revised line first}{}\fi}
\newcommand{\rev@fold}[1]{%
  \rev@read{#1}%
  \expandafter\rev@scan\rev@status; \relax}
% \sectioninput[parts]{file} reads one section file. A section whose
% subsections sit in files of their own lists them as the optional argument:
% the section shows the latest entry across them (Sally, 8 October 2026).
\newcommand{\sectioninput}[2][]{%
  \begingroup
    \endlinechar=-1
    \global\rev@best=0
    \rev@fold{#2}%
    \forcsvlist\rev@fold{#1}%
    \xdef\rev@label{\rev@bestitem}%
  \endgroup
  \input{#2}}
% \partinput[files]{file} is the same, for the file that opens a part: the part
% line stamps the latest entry across the file and the files listed, which are
% the part's own (its sections and their subsections).
\let\partinput\sectioninput
