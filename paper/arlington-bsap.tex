\usepackage{titlesec}
% Heading conventions, set once for the whole report, the Data Appendix included.
% A part (Part A to Part D) opens a page. A section (A.1) is numbered within its
% part and starts a new page. A subsection is unnumbered and italic (Sally,
% 5 October 2026) and runs on; the contents page lists parts and sections.
% Paragraphs are never indented and are
% separated by a line of space (Sally, 4 October 2026), so a heading, a figure
% or a table never changes how one begins.
\usepackage{parskip}
\usepackage{chngcntr}
\titleclass{\part}{straight}
\titleformat{\section}{\normalfont\normalsize\bfseries}{\thesection}{1em}{}
\titleformat{\subsection}{\normalfont\normalsize\itshape}{}{0em}{}
\setcounter{tocdepth}{1}
% The contents page: a part prints as "Part A." the way its heading does, and a
% section's number gets room before its title.
\usepackage{titletoc}
\makeatletter
\renewcommand{\ttl@tocpart}{\def\ttl@a{\protect\numberline{\thepart}}}
\makeatother
\titlecontents{part}[0pt]{\addvspace{10pt}\bfseries}{Part \thecontentslabel.\hspace{0.5em}}{}{\hfill\contentspage}
\titlecontents{section}[2.4em]{}{\contentslabel{2.4em}}{\hspace*{-2.4em}}{\hfill\contentspage}
\renewcommand{\thepart}{\Alph{part}}
\titleformat{\part}{\normalfont\normalsize\bfseries}{Part \thepart.}{0.5em}{}
% One scheme, set once, for every level (Sally, 5 October 2026): one unit,
% a line of body text (\baselineskip), and every heading's space above is
% 1 or 2 of it, never a point value chosen by eye. A subsection and a
% run-in head, which already share the same italic weight, get a line; a
% section and a part, bold, get two, so where sections run on with no page
% break between them (Parts B, C and D), a section still reads as a bigger
% break than a subsection. Three lines read as excessive for a section
% (Sally, 5 October 2026). Space below is one line throughout, so a
% heading sits close to its own text and the hierarchy reads in the space
% above, not below.
\titlespacing*{\part}{0pt}{2\baselineskip}{2\baselineskip}
% Sections are numbered within their part: A.1, A.2, ..., C.1.
\counterwithin*{section}{part}
\renewcommand{\thesection}{\thepart.\arabic{section}}
\titlespacing*{\section}{0pt}{2\baselineskip}{\baselineskip}
\titlespacing*{\subsection}{0pt}{\baselineskip}{\baselineskip}
% Third level, used inside Demographic Representation: unnumbered, italic, run on the
% way a subsection is, a little closer to its text.
\titleformat{\subsubsection}{\normalfont\normalsize\itshape}{}{0em}{}
\titlespacing*{\subsubsection}{0pt}{\baselineskip}{\baselineskip}
